## Score 1

<VexFlow code={`(VexFlow, factory, score) => {
    const system = factory.System();
    const hhhh = () => score.beam(score.notes("G5/8/m, G5/8/m", { stem: "up"}));
    const bd = () => score.notes("F4/q", /*{ stem: "down" }*/);
    const sd = () => score.notes("C5/q", /*{ stem: "down" }*/);
    system.addStave({
        spaceAbove: 1,
        voices: [
          score.voice([hhhh(), hhhh(), hhhh(), hhhh()].flat()),
          score.voice([bd(), sd(), bd(), sd()].flat()),
        ],
      })
      .addClef("percussion")
      .addTimeSignature("4/4");
  }`} />

## Score 2
<VexFlow code={`(VexFlow, factory, score) => {

          score.voice(score.notes("C#5/q, B4, A4, G#4", { stem: "up" })),
          score.voice(score.notes("C#4/h, C#4", { stem: "down" })),
        ],
      })
      .addClef("treble")
      .addTimeSignature("4/4");
}`} />
